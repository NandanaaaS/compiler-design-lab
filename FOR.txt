//WORKS FOR NESTED FOR ALSO

%{
#include "y.tab.h"
%}
%%
"for"                   return FOR;
[a-zA-Z_][a-zA-Z0-9_]*  return ID;
[0-9]+                  return NUM;
"++"                    return INC;
"--"                    return DEC;
"<="                    return LE;
">="                    return GE;
[=<>();]                return yytext[0];
[ \t]                 	;
\n			return 0;
.                       return yytext[0];
%%

--------------------
%{
#include <stdio.h>
%}
%token FOR ID NUM INC DEC LE GE
%%
S: stmt { printf("Valid for statement\n"); };

stmt: loop
    | ID '=' NUM ';'
    | ID '=' ID ';'
    | '{' stmts '}'
    ;

stmts: stmts stmt
     | 
     ;

loop: FOR '(' init ';' cond ';' update ')' stmt;

init: ID '=' NUM | ;
cond: ID '<' NUM | ID '>' NUM | ID LE NUM | ID GE NUM | ;
update: ID INC | ID DEC | ;

%%
int main() { printf("Enter for statement: "); yyparse(); }
int yyerror() { printf("Invalid for statement\n"); return 0; }
int yywrap() { return 1; }
