int yywrap()
{
    return 1;
}

------------------------------

  %{
#include <stdio.h>
int yylex();
int yyerror(char *s);
%}

%token NUM
%left '+' '-'
%left '*' '/'

%%
S : E '\n' { printf("Result = %d\n", $1); return 0; }
  ;

E : E '+' E { $$ = $1 + $3; }
  | E '-' E { $$ = $1 - $3; }
  | E '*' E { $$ = $1 * $3; }
  | E '/' E {
                if($3 == 0)
                {
                    printf("Division by zero\n");
                    return 0;
                }
                $$ = $1 / $3;
             }
  | '(' E ')' { $$ = $2; }
  | NUM       { $$ = $1; }
  ;
%%

int main()
{
    printf("Enter expression: ");
    yyparse();
    return 0;
}

int yyerror(char *s)
{
    printf("Invalid expression\n");
    return 0;
}
